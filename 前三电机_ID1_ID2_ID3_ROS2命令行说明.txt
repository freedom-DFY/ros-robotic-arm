# 前三个电机：Ubuntu 24.04 / ROS 2 Jazzy
# 固定起点后走20cm的当前实机流程，见 calibration/arm3_horizontal_up_20cm/执行20cm实机命令.txt。
# ROS 2 包：src/arm3_ros2；旧三轴 ROS 1 源码目录已删除。
# 详细校准方法见 src/arm3_ros2/README.md。
# 按步骤在对应终端执行，不要把本文件整体当脚本运行。
# 本包仅连接 ID 1、2、3；默认型号 XM540-W270（1120），Protocol 2.0，1 Mbps。
# 不使用 ID 4、5，不能套用双电机说明里的型号编号 1000。
# 关闭 Wizard、五轴节点和 motor_test_node，让本节点独占串口。

# 1. 终端 1：编译并启动无硬件模式。
cd /home/duziteng/ros-robotic-arm
source /opt/ros/jazzy/setup.bash
colcon build --symlink-install --packages-up-to arm3_ros2
source install/setup.bash
ros2 launch arm3_ros2 arm3.launch.py dry_run:=true

# 2. 终端 2：每次新开终端先执行这两条 source。
source /opt/ros/jazzy/setup.bash
source /home/duziteng/ros-robotic-arm/install/setup.bash

# 查看三轴刻度，顺序为 ID 1、2、3。Ctrl+C 退出 echo。
ros2 topic echo /arm3/encoder_ticks
# 查看三关节角度，单位 rad。Ctrl+C 退出 echo。
ros2 topic echo /arm3/joint_states
# 查看原 J4 安装点的位置，单位 m。Ctrl+C 退出 echo。
ros2 topic echo /arm3/wrist_point

# 以下两条运动目标仅用于 dry_run 验证；等待上一条运动完成再发下一条。
ros2 topic pub --once /arm3/joint_target sensor_msgs/msg/JointState \
  '{name: [joint_1, joint_2, joint_3], position: [0.0, 0.18, -0.20]}'
ros2 topic pub --once /arm3/cartesian_target geometry_msgs/msg/PointStamped \
  '{header: {frame_id: base_link}, point: {x: 0.01, y: 0.30, z: 0.15}}'

# 3. 终端 1：Ctrl+C 停止 dry_run，再准备实机配置。
# 支撑大臂、二臂，提前在 Wizard 中关闭三台电机扭矩，然后退出 Wizard。
# Linux 使用 /dev/ttyUSB* 或 /dev/ttyACM*，不能使用 Windows COM5。
cd /home/duziteng/ros-robotic-arm
cp -i src/arm3_ros2/config/arm3.yaml ~/arm3_hardware.yaml
ls -l /dev/serial/by-id/
# 按实际设备路径修改；虚拟机需先完成 USB 直通。
MOTOR_PORT=/dev/ttyUSB0

# 4. 终端 1：先只读连接，不写控制模式、目标或扭矩寄存器。
# dry_run=false 仅表示连接真实串口，不会自动使能电机。
ros2 launch arm3_ros2 arm3.launch.py dry_run:=false \
  "device:=$MOTOR_PORT" baud_rate:=1000000 config:="$HOME/arm3_hardware.yaml"

# 5. 终端 2：核对位置和模型方向。
ros2 topic echo /arm3/encoder_ticks
ros2 topic echo /arm3/joint_states

# 在扭矩关闭时测定每轴 zero_ticks、joint_directions 及机械安全 min/max。
# 编码范围 0~4095 不等于机构行程，必须把每轴限位收紧并留机械余量。
# 软件限位必须在电机内部位置限位之内；不要复制任意姿态读数当机械零位。
# 编辑 ~/arm3_hardware.yaml，保持 ROS 2 的 /**: -> ros__parameters: 层级。
# 校准全部完成后设 calibration_confirmed: true。
# false 或任一轴仍为 0~4095 时仍可使能保持，但运动请求会被拒绝。
# 电机必须已设 Position Control Mode=3、Homing Offset=0、Torque On by Goal Update 关闭。
# 修改配置后，在终端 1 按 Ctrl+C，再执行步骤 4 的 launch 命令重新加载。
# 运行时参数只读；不要在旧节点还运行时启动第二个节点。

# 6. 终端 2：支撑好机构并清空活动范围，手动使能并保持实际位置。
# 使能不检查软件工作空间；发送后续运动前再检查校准、限位和整段路径。
ros2 service call /arm3/enable std_srvs/srv/SetBool '{data: true}'
# 只有 success: true 表示成功。初始目标取自当时实际位置，不会自动回仿真起点。
ros2 topic echo --once /arm3/joint_states

# 完成运动标定后，根据刚读到的角度修改 joint_target：只把一轴加/减约 0.02 rad，另两轴填当前值。
# 下面仅为格式示例：只有当前角度恰好为 [0.0, 0.18, -0.20]，且行程允许时才可照抄。
# position 是三轴绝对目标，不是增量；该例只让 J1 从 0.0 移到 0.02 rad。
ros2 topic pub --once /arm3/joint_target sensor_msgs/msg/JointState \
  '{name: [joint_1, joint_2, joint_3], position: [0.02, 0.18, -0.20]}'
# 单独调试 J2/J3 同理，保留其余两轴当前值；三台电机仍需全部在线。

# 7. 停止并保持；不会主动卸力，也不能代替硬件急停。
ros2 service call /arm3/stop std_srvs/srv/Trigger '{}'
# 支撑机构后卸掉本节点使能的扭矩；失去扭矩后可能因重力下落。
ros2 service call /arm3/enable std_srvs/srv/SetBool '{data: false}'
# 正常 Ctrl+C 退出也会尝试卸力；通信断开时无法保证卸力成功。

# 8. 可选手柄：先在 dry_run 验证映射，需单独安装原 Linux 布局驱动。
sudo apt install ros-jazzy-joy-linux
ros2 launch arm3_ros2 arm3.launch.py dry_run:=true use_joy:=true joy_device:=/dev/input/js0
# BACK 切换；关节模式 LT/RT 调 J1，LB/RB 调 J2，十字键上下调 J3。
# 每次按下走一步，需松开再按；不使用 SDL game_controller_node 的映射。

# 通信故障：先支撑机构并修复供电、USB、线缆、ID/波特率及设备路径。
# 节点故障会清空轨迹并禁止新运动；enable false/true 用于清故障后重新使能。
# USB I/O 异常通常应停止旧节点，重新确认串口后从只读连接开始。
# 仅当报串口权限不足时执行下面命令，之后注销并重新登录。
sudo usermod -aG dialout "$USER"
