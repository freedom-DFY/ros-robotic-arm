cmake_minimum_required(VERSION 3.16)
project(five_dof_arm)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(dynamixel_sdk REQUIRED)
find_package(dynamixel_sdk_custom_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)

add_executable(sync_read_write_node src/sync_read_write_node.cpp)
ament_target_dependencies(sync_read_write_node rclcpp dynamixel_sdk
  dynamixel_sdk_custom_interfaces sensor_msgs)
target_include_directories(sync_read_write_node PRIVATE include)
target_link_libraries(sync_read_write_node Eigen3::Eigen)

add_executable(logicool_controller src/logicool_controller.cpp)
ament_target_dependencies(logicool_controller rclcpp dynamixel_sdk_custom_interfaces
  sensor_msgs geometry_msgs)
target_include_directories(logicool_controller PRIVATE include)
target_link_libraries(logicool_controller Eigen3::Eigen)

install(TARGETS sync_read_write_node logicool_controller DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS src/motor_test_node.py DESTINATION lib/${PROJECT_NAME}
  RENAME motor_test_node)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_kinematics test/test_kinematics.cpp)
  target_include_directories(test_kinematics PRIVATE include)
  target_link_libraries(test_kinematics Eigen3::Eigen)
  ament_add_pytest_test(test_motor_test_node test/test_motor_test_node.py)
endif()
ament_package()
